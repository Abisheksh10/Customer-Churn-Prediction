CREATE TABLE customers1 (
    customer_id VARCHAR(20) PRIMARY KEY,
    gender VARCHAR(10),
    senior_citizen INT,
    partner INT,
    dependents INT
);
INSERT INTO customers1 (
    customer_id,
    gender,
    senior_citizen,
    partner,
    dependents
)
SELECT
    customerID,
    LOWER(gender),
    CAST(SeniorCitizen AS UNSIGNED),

    CASE
        WHEN Partner = 'Yes' THEN 1
        WHEN Partner = 'No' THEN 0
        ELSE NULL
    END,

    CASE
        WHEN Dependents = 'Yes' THEN 1
        WHEN Dependents = 'No' THEN 0
        ELSE NULL
    END

FROM stg_customer_raw;







-- 62
CREATE TABLE fact_customer_account (
    customer_id VARCHAR(20) PRIMARY KEY,
    tenure INT,
    monthly_charges DECIMAL(10,2),
    total_charges DECIMAL(12,2),
    churn INT,
    contract_id INT,
    payment_id INT,

    FOREIGN KEY (customer_id)
        REFERENCES customers1(customer_id),

    FOREIGN KEY (contract_id)
        REFERENCES dim_contract(contract_id),

    FOREIGN KEY (payment_id)
        REFERENCES dim_payment(payment_id)
);

-- the error i got is because of the empty space. so trim it.
INSERT INTO fact_customer_account (
    customer_id,
    tenure,
    monthly_charges,
    total_charges,
    churn,
    contract_id,
    payment_id
)
SELECT
    s.customerID,

    s.tenure,

    s.MonthlyCharges,

    CAST(
        NULLIF(TRIM(s.TotalCharges), '')
        AS DECIMAL(12,2)
    ),

    CASE
        WHEN s.Churn = 'Yes' THEN 1
        WHEN s.Churn = 'No' THEN 0
        ELSE NULL
    END,

    c.contract_id,

    p.payment_id

FROM stg_customer_raw s

JOIN dim_contract c
    ON s.Contract = c.contract_name

JOIN dim_payment p
    ON s.PaymentMethod = p.payment_method;
