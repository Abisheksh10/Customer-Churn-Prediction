use telecom;
CREATE TABLE customer_ml_features (
    customer_id VARCHAR(20) PRIMARY KEY,
    tenure INT NOT NULL,
    monthly_charges FLOAT NOT NULL,
    total_charges FLOAT,
    tenure_bucket VARCHAR(10) NOT NULL,
    high_charge_flag INT NOT NULL,
    service_count INT NOT NULL,
    is_long_term_customer INT NOT NULL,
    has_streaming_bundle INT NOT NULL,
    auto_pay_flag INT NOT NULL,
    churn INT NOT NULL
);
INSERT INTO customer_ml_features (
    customer_id,
    tenure,
    monthly_charges,
    total_charges,
    tenure_bucket,
    high_charge_flag,
    service_count,
    is_long_term_customer,
    has_streaming_bundle,
    auto_pay_flag,
    churn
)
SELECT
    f.customer_id,

    f.tenure,

    f.monthly_charges,

    COALESCE(
        f.total_charges,
        f.monthly_charges
    ),

    CASE
        WHEN f.tenure <= 12 THEN '0-12'
        WHEN f.tenure <= 24 THEN '13-24'
        WHEN f.tenure <= 48 THEN '25-48'
        ELSE '49-72'
    END,

    CASE
        WHEN f.monthly_charges > (
            SELECT AVG(monthly_charges)
            FROM fact_customer_account
        )
        THEN 1
        ELSE 0
    END,

    (
        CASE WHEN s.OnlineSecurity = 'Yes' THEN 1 ELSE 0 END
        +
        CASE WHEN s.OnlineBackup = 'Yes' THEN 1 ELSE 0 END
        +
        CASE WHEN s.DeviceProtection = 'Yes' THEN 1 ELSE 0 END
        +
        CASE WHEN s.TechSupport = 'Yes' THEN 1 ELSE 0 END
        +
        CASE WHEN s.StreamingTV = 'Yes' THEN 1 ELSE 0 END
        +
        CASE WHEN s.StreamingMovies = 'Yes' THEN 1 ELSE 0 END
    ),

    CASE
        WHEN f.tenure >= 24 THEN 1
        ELSE 0
    END,

    CASE
        WHEN s.StreamingTV = 'Yes'
         AND s.StreamingMovies = 'Yes'
        THEN 1
        ELSE 0
    END,

    CASE
        WHEN s.PaymentMethod LIKE '%automatic%'
        THEN 1
        ELSE 0
    END,

    f.churn

FROM fact_customer_account f

JOIN stg_customer_raw s
    ON f.customer_id = s.customerID;

SELECT COUNT(*) AS null_monthly_charges
FROM customer_ml_features
WHERE monthly_charges IS NULL;

SELECT
    COUNT(*) AS total_rows,
    COUNT(*) - COUNT(tenure) AS null_tenure,
    COUNT(*) - COUNT(monthly_charges) AS null_monthly_charges,
    COUNT(*) - COUNT(total_charges) AS null_total_charges,
    COUNT(*) - COUNT(tenure_bucket) AS null_tenure_bucket,
    COUNT(*) - COUNT(high_charge_flag) AS null_high_charge_flag,
    COUNT(*) - COUNT(service_count) AS null_service_count,
    COUNT(*) - COUNT(is_long_term_customer) AS null_long_term,
    COUNT(*) - COUNT(has_streaming_bundle) AS null_streaming_bundle,
    COUNT(*) - COUNT(auto_pay_flag) AS null_auto_pay,
    COUNT(*) - COUNT(churn) AS null_churn
FROM customer_ml_features;

SELECT
    churn,
    COUNT(*) AS customer_count
FROM customer_ml_features
GROUP BY churn
ORDER BY churn;
